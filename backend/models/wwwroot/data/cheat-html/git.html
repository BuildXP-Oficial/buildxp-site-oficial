SETUP & CONFIG
git --versionVer versão do Git instalada
git config --global user.name "Nome"Definir nome do usuário
git config --global user.email "email"Definir e-mail do usuário
git config --listListar todas as configurações
git config --global init.defaultBranch mainNovos repos nascem com branch main
REPOSITÓRIO
git initInicializar repositório na pasta atual
git clone <url>Clonar repositório remoto
git clone --depth 1 <url>Clone raso (só último commit, mais rápido)
git remote -vListar repositórios remotos
git remote add origin <url>Conectar ao repositório remoto
git remote set-url origin <url>Trocar URL do origin (HTTPS/SSH)
git fetch originBaixar refs/commits sem integrar
git fetch --prune originFetch + remover refs de branches apagadas no remoto
GITHUB CLI (gh)
gh auth loginLogin interativo no GitHub
gh repo create <nome> --public --source=. --remote=origin --pushCriar repo no GitHub e dar push da pasta atual
gh pr createAbrir PR pelo terminal (fluxo interativo)
gh pr checkout <número>Checkout da branch do PR localmente
STAGE & COMMIT
git statusVer estado dos arquivos
git add .Stage de todos os arquivos modificados
git add <arquivo>Stage de arquivo específico
git commit -m "msg"Commitar com mensagem
git commit -am "msg"Stage + commit só em arquivos já rastreados
git commit --amendEditar o último commit
git commit --amend --no-editIncluir mudanças no último commit (mesma msg)
git log --onelineHistórico resumido de commits
git log --graph --oneline --all -15Árvore de branches + últimos commits
git show <hash>Ver patch e metadados de um commit
git diffVer mudanças não staged
git diff --stagedDiff do que está no stage (antes do commit)
git blame <arquivo>Quem mudou cada linha (e em qual commit)
BRANCHES
git branchListar branches locais
git checkout -b <nome>Criar e entrar na branch
git switch -c <nome>Criar e entrar (equivalente moderno ao checkout -b)
git switch <nome>Trocar de branch (Git moderno)
git branch -aListar branches locais + remotas
git merge <branch>Mergear branch na atual
git merge --no-ff <branch>Merge sempre com commit de merge (histórico explícito)
git rebase <branch>Reaplicar commits da branch atual em cima da base
git rebase -i HEAD~3Rebase interativo dos últimos 3 commits (squash/reorder)
git cherry-pick <hash>Trazer um commit específico para a branch atual
git branch -d <nome>Deletar branch local
git branch -D <nome>Forçar delete de branch local não mergeada
PUSH & PULL
git pushEnviar commits ao remoto
git push -u origin <branch>Push inicial com upstream
git pullPuxar e mergear mudanças
git pull --rebasePuxar reaplicando seus commits em cima do remoto
git fetchPuxar sem mergear
git push origin --delete <branch>Deletar branch remota
TAGS & SUBMÓDULOS
git tag -a v1.0 -m "release"Tag anotada (releases/versionamento)
git push origin v1.0Enviar uma tag ao remoto
git push origin --tagsEnviar todas as tags locais
git submodule update --init --recursiveBaixar/atualizar submódulos após clone
DESFAZER & EMERGÊNCIA
git restore <arquivo>Descartar mudanças não commitadas
git restore --staged <arquivo>Remover do stage
git reset --soft HEAD~1Desfazer último commit (mantém mudanças)
git reset --hard HEAD~1Desfazer e APAGAR mudanças ⚠️
git stashSalvar mudanças temporariamente
git stash push -u -m "wip"Stash incluindo arquivos novos não rastreados
git stash listListar entradas do stash
git stash apply stash@{0}Aplicar stash sem remover da pilha
git stash popRestaurar do stash e remover o topo
git revert <hash>Reverter commit sem apagar histórico
git reflogHistórico de onde HEAD esteve (recuperar commits)
git clean -fdApagar arquivos não rastreados ⚠️ irreversível
DEBUG & INVESTIGAÇÃO
git bisect startIniciar busca binária do commit que quebrou algo
git bisect badMarcar commit atual como ruim
git bisect good <hash>Marcar um commit conhecido como bom
git bisect resetSair do bisect e voltar ao estado anterior
