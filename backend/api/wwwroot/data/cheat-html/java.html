INSTALAÇÃO & VERSÃO
java -versionVer versão do JRE/JDK em uso
javac -versionVer versão do compilador
echo %JAVA_HOME%Ver JAVA_HOME (Windows CMD)
echo $JAVA_HOMEVer JAVA_HOME (Linux/macOS)
where javaCaminho do executável (Windows)
which javaCaminho do executável (Unix)
COMPILAR & EXECUTAR
javac Main.javaCompilar um ficheiro .java
javac -d out src/**/*.javaCompilar vários ficheiros para pasta out
java MainExecutar classe com método main
java -cp out com.app.MainExecutar com classpath explícito
java -jar app.jarExecutar JAR empacotado
java -jar app.jar --spring.profiles.active=devPassar argumentos ao JAR (ex.: Spring)
JAR & MANIFEST
jar cf app.jar -C out .Criar JAR a partir de classes compiladas
jar tf app.jarListar conteúdo do JAR
jar xf app.jarExtrair ficheiros do JAR
jar cfe app.jar com.app.Main -C out .JAR executável com Main-Class no manifest
javadoc -d docs src/**/*.javaGerar documentação HTML do código
MAVEN — PROJETO
mvn -versionVer versão do Maven e Java
mvn archetype:generateCriar projeto a partir de archetype
mvn clean compileLimpar e compilar
mvn clean packageCompilar e gerar JAR/WAR
mvn clean installBuild + instalar no repositório local
mvn clean verifyBuild completo com testes de integração
mvn dependency:treeÁrvore de dependências
mvn versions:display-dependency-updatesVer dependências desatualizadas
MAVEN — EXECUTAR & SPRING
mvn exec:java -Dexec.mainClass="com.app.Main"Executar classe main via Maven
mvn spring-boot:runSubir app Spring Boot em dev
mvn spring-boot:build-imageGerar imagem OCI com Cloud Native Buildpacks
mvn -DskipTests packageEmpacotar sem correr testes
mvn -pl modulo-a -am testTestar módulo específico num multi-módulo
GRADLE
gradle -vVer versão do Gradle
gradle initInicializar projeto Gradle
./gradlew buildCompilar e testar (wrapper Unix)
gradlew.bat buildCompilar e testar (wrapper Windows)
./gradlew bootRunExecutar Spring Boot via Gradle
./gradlew clean testLimpar e correr testes
./gradlew dependenciesListar dependências do projeto
TESTES
mvn testCorrer testes JUnit com Maven
mvn -Dtest=NomeDaClasseTest testCorrer uma classe de teste
mvn -Dtest=NomeDaClasseTest#metodo testCorrer um método de teste
./gradlew testCorrer testes com Gradle
./gradlew test --tests "com.app.*"Filtrar testes por pacote/classe
MÓDULOS (JPMS)
javac --module-path lib -d out -m com.app/com.app.MainCompilar módulo com module-path
java --module-path out --module com.app/com.app.MainExecutar módulo nomeado
jar --create --file app.jar --main-class com.app.Main -C out .Criar JAR modular
DEBUG & DIAGNÓSTICO
java -XshowSettings:properties -versionVer propriedades da JVM
jps -lListar processos Java em execução
jcmd <pid> VM.flagsFlags da JVM de um processo
jstack <pid>Dump de threads (deadlock/debug)
java -agentlib:jdwp=transport=dt_socket,server=y,suspend=n,address=*:5005 -jar app.jarAtivar debug remoto na porta 5005
