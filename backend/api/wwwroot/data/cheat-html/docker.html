IMAGENS
docker pull <imagem>Baixar imagem do Docker Hub
docker imagesListar imagens locais
docker build -t <nome> .Buildar imagem do Dockerfile atual
docker rmi <imagem>Remover imagem local
docker image pruneRemover imagens sem uso
CONTAINERS
docker run <imagem>Criar e iniciar container
docker run -d -p 8080:80 <imagem>Rodar em background com porta mapeada
docker run -it <imagem> bashEntrar no container interativamente
docker psListar containers rodando
docker ps -aListar todos os containers
docker stop <container>Parar container
docker start <container>Iniciar container parado
docker restart <container>Reiniciar container
docker rm <container>Remover container
docker logs <container>Ver logs do container
docker exec -it <container> bashAbrir terminal no container rodando
docker inspect <container>Ver detalhes do container
VOLUMES & REDES
docker volume create <nome>Criar volume persistente
docker volume lsListar volumes
docker run -v <vol>:/data <img>Montar volume no container
docker network lsListar redes
docker network create <nome>Criar rede customizada
DOCKER COMPOSE
docker compose upSubir todos os serviços
docker compose up -dSubir em background
docker compose downParar e remover containers
docker compose logsVer logs de todos os serviços
docker compose buildRebuildar imagens do compose
docker compose psVer status dos serviços
LIMPEZA
docker system pruneRemover tudo que não está em uso
docker container pruneRemover containers parados
docker image prune -aRemover todas as imagens sem uso
